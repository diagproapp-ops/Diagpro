def get_db_connection(db_path: str) -> sqlite3.Connection:
    """Estabelece e retorna uma conexão com o banco de dados SQLite."""
    # O logger global será usado aqui se o logger do módulo não for configurado antes
    global_logger = logging.getLogger(__name__) # Logger para esta função específica
    try:
        conn = sqlite3.connect(db_path)
        conn.row_factory = sqlite3.Row  # Acessar colunas por nome
        conn.execute("PRAGMA foreign_keys = ON;") # Habilitar chaves estrangeiras
        return conn
    except sqlite3.Error as e:
        global_logger.error(f"Erro ao conectar ao banco de dados '{db_path}': {e}", exc_info=True)
        raise

class EventManager:
    def __init__(self, db_path: str, app_controller: Optional[Any] = None, logger: Optional[logging.Logger] = None):
        self.db_path = db_path
        self.app_controller = app_controller # Armazena a referência se necessário para callbacks, etc.
        
        # Usa o logger injetado, ou cria um novo se não for fornecido
        if logger:
            self.logger = logger
        else:
            self.logger = logging.getLogger(__name__ + ".EventManagerInstance") # Nome mais específico
            if not self.logger.hasHandlers(): # Configura um handler básico se nenhum existir
                logging.basicConfig(level=logging.INFO)
                self.logger.info("Logger do EventManager configurado com basicConfig pois nenhum logger foi injetado ou configurado globalmente.")

        if not os.path.exists(self.db_path):
            self.logger.warning(f"Arquivo de banco de dados '{self.db_path}' não encontrado. Será criado na primeira conexão.")
            db_dir = os.path.dirname(self.db_path)
            if db_dir and not os.path.exists(db_dir):
                try:
                    os.makedirs(db_dir, exist_ok=True)
                    self.logger.info(f"Diretório para o banco de dados criado: {db_dir}")
                except OSError as e:
                    self.logger.error(f"Não foi possível criar o diretório para o banco de dados '{db_dir}': {e}")

        try:
            from .database_setup import create_all_tables_if_not_exists
            create_all_tables_if_not_exists() 
            self.logger.info("Rotina de criação/verificação de tabelas executada com sucesso via EventManager.")
        except ImportError:
            self.logger.error("Falha ao importar 'database_setup'. As tabelas podem não ser criadas/verificadas.")
        except Exception as e_db_setup:
            self.logger.error(f"Erro ao executar database_setup: {e_db_setup}", exc_info=True)

        try:
            with get_db_connection(self.db_path) as conn:
                self.logger.info(f"EventManager: Conectado ao banco de dados: {self.db_path}")
        except sqlite3.Error as e:
            self.logger.error(f"EventManager: Falha na conexão inicial com o banco de dados: {e}", exc_info=True)


    def _execute_query(self, query: str, params: tuple = (), fetch_one: bool = False, fetch_all: bool = False, commit: bool = False) -> Optional[Union[sqlite3.Row, List[sqlite3.Row], int]]:
        """
        Executa uma query SQL.
        Retorna um único resultado, todos os resultados, o ID da linha inserida, ou None.
        """
        conn = None
        try:
            conn = get_db_connection(self.db_path)
            cursor = conn.cursor()
            cursor.execute(query, params)

            if commit:
                conn.commit()
                return cursor.lastrowid
            
            if fetch_one:
                return cursor.fetchone()
            if fetch_all:
                return cursor.fetchall()
            return None 
        except sqlite3.Error as e:
            error_msg = f"Erro SQLite ao executar query:\nQuery:\n{query}\nParams: {params}\nErro: {e}"
            self.logger.error(error_msg, exc_info=True) # Usa o logger da instância
            return None 
        finally:
            if conn:
                conn.close()

    def create_event(self, evento_data: Union[EventoType, Dict[str, Any]]) -> Optional[str]: 
        """Cria um novo evento no banco de dados."""
        self.logger.info(f"Criando novo evento: {evento_data.get('nome_evento') if isinstance(evento_data, dict) else evento_data.nome_evento}")

        if isinstance(evento_data, dict):
            event_id = evento_data.get('id_evento', str(uuid.uuid4()))
            nome_evento = evento_data.get('nome_evento')
            data_prova = evento_data.get('data_prova')
            numero_edital = evento_data.get('numero_edital')
            local_prova = evento_data.get('local_prova')
            instituicao_organizadora = evento_data.get('instituicao_organizadora')
            logo_path = evento_data.get('logo_path')
            observacoes = evento_data.get('observacoes')
            status = evento_data.get('status', 'Planejado')
            qtd_cargos_estimada = evento_data.get('qtd_cargos_estimada', 0)
        else: # Assume EventoType
            event_id = evento_data.id_evento
            nome_evento = evento_data.nome_evento
            data_prova = str(evento_data.data_prova) if evento_data.data_prova else None
            numero_edital = evento_data.numero_edital
            local_prova = evento_data.local_prova
            instituicao_organizadora = evento_data.instituicao_organizadora
            logo_path = evento_data.logo_path
            observacoes = evento_data.observacoes
            status = evento_data.status
            qtd_cargos_estimada = evento_data.qtd_cargos_estimada
            
        sql = """
            INSERT INTO Eventos (
                id_evento, nome_evento, data_prova, numero_edital, local_prova, 
                instituicao_organizadora, logo_path, observacoes, status, qtd_cargos_estimada
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """
        params = (
            event_id, nome_evento, data_prova, numero_edital, local_prova,
            instituicao_organizadora, logo_path, observacoes, status, qtd_cargos_estimada
        )
        
        inserted_row_id = self._execute_query(sql, params, commit=True)
        if inserted_row_id is not None: 
            self.logger.info(f"Evento '{nome_evento}' (ID: {event_id}) criado com sucesso.")
            return event_id
        else:
            self.logger.error(f"Falha ao criar evento '{nome_evento}'.")
            return None

    def get_event_by_id(self, event_id: str) -> Optional[Dict[str, Any]]:
        """Busca um evento pelo ID."""
        sql = "SELECT * FROM Eventos WHERE id_evento = ?"
        row = self._execute_query(sql, (event_id,), fetch_one=True)
        if row:
            evento_dict = dict(row)
            
            sql_cargos = "SELECT * FROM Cargos WHERE id_evento_fk = ? ORDER BY nome_cargo;"
            cargos_rows = self._execute_query(sql_cargos, (event_id,), fetch_all=True)

            cargos_list = []
            if cargos_rows:
                for cargo_row_dict in [dict(cr) for cr in cargos_rows]: # Converte cada linha para dict primeiro
                    # Desserializar campos JSON para CargoType
                    if 'disciplinas_prova' in cargo_row_dict and isinstance(cargo_row_dict['disciplinas_prova'], str):
                        try:
                            cargo_row_dict['disciplinas_prova'] = json.loads(cargo_row_dict['disciplinas_prova'])
                        except json.JSONDecodeError:
                            self.logger.error(f"Erro ao desserializar 'disciplinas_prova' para o cargo ID {cargo_row_dict.get('id_cargo')}: {cargo_row_dict['disciplinas_prova']}")
                            cargo_row_dict['disciplinas_prova'] = [] # Ou outro valor padrão apropriado

                    if 'cover_pdf_paths' in cargo_row_dict and isinstance(cargo_row_dict['cover_pdf_paths'], str):
                        try:
                            cargo_row_dict['cover_pdf_paths'] = json.loads(cargo_row_dict['cover_pdf_paths'])
                        except json.JSONDecodeError:
                            self.logger.error(f"Erro ao desserializar 'cover_pdf_paths' para o cargo ID {cargo_row_dict.get('id_cargo')}: {cargo_row_dict['cover_pdf_paths']}")
                            cargo_row_dict['cover_pdf_paths'] = []

                    if 'metadados_cargo' in cargo_row_dict and isinstance(cargo_row_dict['metadados_cargo'], str):
                        try:
                            cargo_row_dict['metadados_cargo'] = json.loads(cargo_row_dict['metadados_cargo'])
                        except json.JSONDecodeError:
                            self.logger.error(f"Erro ao desserializar 'metadados_cargo' para o cargo ID {cargo_row_dict.get('id_cargo')}: {cargo_row_dict['metadados_cargo']}")
                            cargo_row_dict['metadados_cargo'] = {}

                    # Adicione desserialização similar para outros campos JSON em CargoType se houver
                    cargos_list.append(cargo_row_dict)

            evento_dict['cargos'] = cargos_list

            # Faça o mesmo para 'textos_referencia' e 'provas_montadas' se eles contiverem campos JSON stringificados
            # Exemplo para provas_montadas, se 'questoes_por_disciplina' ou 'metadados_prova' forem JSON strings:
            sql_provas = "SELECT * FROM ProvasMontadas WHERE id_evento_fk = ? ORDER BY id_cargo_fk, nome_prova_montada;"
            provas_rows = self._execute_query(sql_provas, (event_id,), fetch_all=True)
            provas_list = []
            if provas_rows:
                for prova_row_dict in [dict(pr) for pr in provas_rows]:
                    if 'questoes_por_disciplina' in prova_row_dict and isinstance(prova_row_dict['questoes_por_disciplina'], str):
                        try:
                            prova_row_dict['questoes_por_disciplina'] = json.loads(prova_row_dict['questoes_por_disciplina'])
                        except json.JSONDecodeError:
                            prova_row_dict['questoes_por_disciplina'] = {}

                    if 'metadados_prova' in prova_row_dict and isinstance(prova_row_dict['metadados_prova'], str):
                        try:
                            prova_row_dict['metadados_prova'] = json.loads(prova_row_dict['metadados_prova'])
                        except json.JSONDecodeError:
                            prova_row_dict['metadados_prova'] = {}

                    if 'ordem_disciplinas_na_prova' in prova_row_dict and isinstance(prova_row_dict['ordem_disciplinas_na_prova'], str):
                        try:
                            prova_row_dict['ordem_disciplinas_na_prova'] = json.loads(prova_row_dict['ordem_disciplinas_na_prova'])
                        except json.JSONDecodeError:
                            prova_row_dict['ordem_disciplinas_na_prova'] = []
                    provas_list.append(prova_row_dict)
            evento_dict['provas_montadas_por_cargo'] = {p['id_cargo_fk']: p for p in provas_list} # Ajuste para o formato esperado por EventoType

            return evento_dict
        return None

    def get_all_events_summary(self) -> List[Dict[str, Any]]:
        """Retorna uma lista de resumos de todos os eventos."""
        sql = """
            SELECT 
                e.id_evento, 
                e.nome_evento, 
                e.data_prova, 
                e.status,
                (SELECT COUNT(*) FROM Cargos c WHERE c.id_evento_fk = e.id_evento) as numero_cargos,
                (SELECT COUNT(*) FROM ProvasMontadas pm WHERE pm.id_evento_fk = e.id_evento) as numero_provas_montadas
            FROM Eventos e
            ORDER BY e.data_modificacao DESC;
        """ 
        rows = self._execute_query(sql, fetch_all=True)
        return [dict(row) for row in rows] if rows else []

    def _cargo_to_dict_for_update(self, cargo_obj: CargoType) -> Dict[str, Any]:
        """Converte um objeto CargoType em um dicionário para UPDATE, serializando JSON."""
        cargo_dict = cargo_obj.model_dump(exclude={'id_evento', 'id_cargo'}) # Exclui chaves que não devem ser atualizadas diretamente ou não são colunas

        # Serializa campos que são listas/dicionários para strings JSON
        # (Similar ao que fizemos para add_cargo_to_event ou _cargo_to_dict)
        for field_name, field_value in cargo_dict.items():
            if isinstance(field_value, (list, dict)):
                try:
                    cargo_dict[field_name] = json.dumps(field_value)
                except TypeError:
                    self.logger.warning(f"Não foi possível serializar o campo '{field_name}' para JSON no cargo ID {getattr(cargo_obj, 'id_cargo', 'N/A')}")
                    # Decida um valor padrão ou como lidar com isso
                    if isinstance(field_value, list): cargo_dict[field_name] = "[]"
                    elif isinstance(field_value, dict): cargo_dict[field_name] = "{}"
        return cargo_dict

    def update_cargo_in_event(self, id_evento: str, cargo_to_update: CargoType) -> bool:
        """
        Atualiza um cargo existente no evento.
        O objeto cargo_to_update já deve ter seu id_cargo definido.
        """
        if not cargo_to_update.id_cargo:
            self.logger.error("Tentativa de atualizar cargo sem id_cargo.")
            return False

        # Converte o objeto CargoType para um dicionário, serializando campos JSON
        cargo_dict_for_db = self._cargo_to_dict_for_update(cargo_to_update)

        # Remove campos que não devem ser atualizados ou não existem na tabela (como o ID do evento se não for FK direta)
        # ou campos de data de criação, etc.
        # id_evento_fk = cargo_dict_for_db.pop('id_evento_fk', id_evento) # Se o campo é id_evento_fk

        fields = []
        values = []
        for key, value in cargo_dict_for_db.items():
            # Adicione aqui uma verificação se 'key' é uma coluna válida da tabela Cargos
            # para evitar erros de SQL. Ex: if key in VALID_CARGO_COLUMNS_FOR_UPDATE:
            fields.append(f"{key} = ?")
            values.append(value)

        if not fields:
            self.logger.warning(f"Nenhum campo para atualizar para o cargo ID {cargo_to_update.id_cargo}.")
            return False # Ou True, se considerar que "nada a fazer" é um sucesso.

        values.append(cargo_to_update.id_cargo) # Para a cláusula WHERE
        values.append(id_evento) # Para a cláusula WHERE (garante que o cargo pertence ao evento)

        sql = f"UPDATE Cargos SET {', '.join(fields)} WHERE id_cargo = ? AND id_evento_fk = ?"

        try:
            # self.logger.debug(f"Executando SQL para atualizar cargo: {sql} com valores: {values}")
            rows_affected = self._execute_query(sql, tuple(values), fetch_all=False, commit=True, is_update_delete=True)
            if rows_affected is not None and rows_affected > 0:
                self.logger.info(f"Cargo ID {cargo_to_update.id_cargo} atualizado com sucesso no evento ID {id_evento}.")
                # Atualizar data_modificacao do evento pai
                self.update_event_modification_timestamp(id_evento)
                return True
            elif rows_affected == 0:
                self.logger.warning(f"Nenhum cargo encontrado com ID {cargo_to_update.id_cargo} no evento {id_evento} para atualizar, ou os dados eram os mesmos.")
                return False # Ou True, dependendo se "nenhuma mudança" é erro.
            else: # rows_affected is None (erro na execução)
                self.logger.error(f"Falha ao atualizar o cargo ID {cargo_to_update.id_cargo}. _execute_query retornou None.")
                return False
        except Exception as e:
            self.logger.error(f"Exceção ao atualizar cargo ID {cargo_to_update.id_cargo}: {e}", exc_info=True)
            return False

    def delete_questao_banco(self, id_questao: str) -> bool:
        """Deleta uma questão do banco e suas opções associadas (via ON DELETE CASCADE)."""
        sql = "DELETE FROM QuestoesBanco WHERE id_questao = ?"
        conn = get_db_connection(self.db_path)
        cursor = conn.cursor()
        try:
            cursor.execute(sql, (id_questao,))
            conn.commit()
            if cursor.rowcount > 0:
                self.logger.info(f"Questão ID {id_questao} e suas opções deletadas com sucesso.")
                return True
            else:
                self.logger.warning(f"Questão ID {id_questao} não encontrada para deleção.")
                return False
        except sqlite3.Error as e_delete:
            self.logger.error(f"Falha ao deletar questão ID {id_questao}: {e_delete}", exc_info=True)
            return False
        finally:
            if conn:
                conn.close()

    # --- Métodos para Provas Montadas ---
    def save_prova_montada(self, prova_data: Union[ProvaMontadaType, Dict[str, Any]]) -> Optional[str]: 
        """Salva (cria ou atualiza) uma ProvaMontada."""
        if isinstance(prova_data, dict):
            pm_id = prova_data.get('id_prova_montada', str(uuid.uuid4()))
        else: # Assume ProvaMontadaType
            pm_id = prova_data.id_prova_montada

        sql = """
            INSERT OR REPLACE INTO ProvasMontadas (
                id_prova_montada, id_evento_fk, id_cargo_fk, nome_prova_montada, identificador_prova,
                titulo_prova, instrucoes_gerais, texto_cabecalho_prova_docx, texto_rodape_prova_docx,
                frase_assinatura_folha_rosto, status_montagem, responsavel_montagem,
                data_ultima_geracao, caminho_arquivo_gerado, hash_prova_gerada, configuracoes_prova,
                data_modificacao 
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP) 
        """
        if isinstance(prova_data, dict):
            config_prova_json = json.dumps(prova_data.get('configuracoes', {})) 
            params = (
                pm_id, prova_data.get('id_evento_fk'), prova_data.get('id_cargo_fk'), 
                prova_data.get('nome_prova_montada'), prova_data.get('identificador_prova'),
                prova_data.get('titulo_prova'), prova_data.get('instrucoes_gerais'),
                prova_data.get('texto_cabecalho_prova_docx'), prova_data.get('texto_rodape_prova_docx'),
                prova_data.get('frase_assinatura_folha_rosto'), prova_data.get('status_montagem', 'Em Elaboração'),
                prova_data.get('responsavel_montagem'), prova_data.get('data_ultima_geracao'),
                prova_data.get('caminho_arquivo_gerado'), prova_data.get('hash_prova_gerada'),
                config_prova_json
            )
        else: # ProvaMontadaType
            config_prova_json = json.dumps(prova_data.configuracoes.__dict__ if hasattr(prova_data.configuracoes, '__dict__') else {})
            params = (
                prova_data.id_prova_montada, prova_data.id_evento_fk, prova_data.id_cargo_fk, 
                prova_data.nome_prova_montada, prova_data.identificador_prova,
                prova_data.titulo_prova, prova_data.instrucoes_gerais,
                prova_data.texto_cabecalho_prova_docx, prova_data.texto_rodape_prova_docx,
                prova_data.frase_assinatura_folha_rosto, prova_data.status_montagem,
                prova_data.responsavel_montagem, str(prova_data.data_ultima_geracao) if prova_data.data_ultima_geracao else None,
                prova_data.caminho_arquivo_gerado, prova_data.hash_prova_gerada,
                config_prova_json
            )
        
        result = self._execute_query(sql, params, commit=True)
        if result is not None:
            self.logger.info(f"Prova Montada ID {pm_id} salva com sucesso.")
            return pm_id
        else:
            self.logger.error(f"Falha ao salvar Prova Montada ID {pm_id}.")
            return None

    def get_provas_montadas_by_cargo_id(self, cargo_id: str) -> List[Dict[str, Any]]:
        """Busca todas as provas montadas para um cargo específico."""
        sql = "SELECT * FROM ProvasMontadas WHERE id_cargo_fk = ? ORDER BY nome_prova_montada, identificador_prova;"
        rows = self._execute_query(sql, (cargo_id,), fetch_all=True)
        provas = []
        if rows:
            for row_data in rows:
                prova_dict = dict(row_data)
                try:
                    if prova_dict.get('configuracoes_prova'):
                        prova_dict['configuracoes'] = json.loads(prova_dict['configuracoes_prova'])
                    else:
                        prova_dict['configuracoes'] = {}
                except json.JSONDecodeError:
                    self.logger.warning(f"Falha ao desserializar configuracoes_prova para Prova ID {prova_dict.get('id_prova_montada')}")
                    prova_dict['configuracoes'] = {}
                provas.append(prova_dict)
        return provas

    def get_prova_montada_by_id(self, id_prova_montada: str) -> Optional[Dict[str, Any]]:
        """Busca uma prova montada específica pelo seu ID, incluindo sua estrutura."""
        sql = "SELECT * FROM ProvasMontadas WHERE id_prova_montada = ?;"
        row = self._execute_query(sql, (id_prova_montada,), fetch_one=True)
        if row:
            prova_dict = dict(row)
            try:
                if prova_dict.get('configuracoes_prova'):
                    prova_dict['configuracoes'] = json.loads(prova_dict['configuracoes_prova'])
                else:
                    prova_dict['configuracoes'] = {} 
            except json.JSONDecodeError:
                self.logger.warning(f"Falha ao desserializar configuracoes_prova para Prova ID {id_prova_montada}")
                prova_dict['configuracoes'] = {}
            
            sql_blocos = "SELECT * FROM BlocosQuestoesProva WHERE id_prova_montada_fk = ? ORDER BY ordem_bloco;"
            blocos_rows = self._execute_query(sql_blocos, (id_prova_montada,), fetch_all=True)
            prova_dict['blocos_questoes'] = []
            if blocos_rows:
                for bloco_row_data in blocos_rows:
                    bloco_dict = dict(bloco_row_data)
                    sql_questoes_bloco = """
                        SELECT qbp.*, qb.enunciado, qb.tipo, qb.disciplina_banco 
                        FROM QuestoesPorBlocoProva qbp
                        JOIN QuestoesBanco qb ON qbp.id_questao_banco_fk = qb.id_questao
                        WHERE qbp.id_bloco_prova_fk = ? 
                        ORDER BY qbp.numero_questao_na_prova;
                    """ 
                    questoes_bloco_rows = self._execute_query(sql_questoes_bloco, (bloco_dict['id_bloco_prova'],), fetch_all=True)
                    bloco_dict['questoes'] = [dict(q_b_row) for q_b_row in questoes_bloco_rows] if questoes_bloco_rows else []
                    prova_dict['blocos_questoes'].append(bloco_dict)
            return prova_dict
        return None

    def delete_prova_montada(self, id_prova_montada: str) -> bool:
        """Deleta uma prova montada e sua estrutura associada (blocos, questões no bloco)."""
        sql = "DELETE FROM ProvasMontadas WHERE id_prova_montada = ?"
        conn = get_db_connection(self.db_path)
        cursor = conn.cursor()
        try:
            cursor.execute(sql, (id_prova_montada,)) 
            conn.commit()
            if cursor.rowcount > 0:
                self.logger.info(f"Prova Montada ID {id_prova_montada} deletada com sucesso.")
                return True
            else:
                self.logger.warning(f"Prova Montada ID {id_prova_montada} não encontrada para deleção.")
                return False
        except sqlite3.Error as e_delete:
            self.logger.error(f"Falha ao deletar Prova Montada ID {id_prova_montada}: {e_delete}", exc_info=True)
            return False
        finally:
            if conn:
                conn.close()

    def get_disciplinas_from_event_question_bank(self, event_id: str) -> List[str]:
        """
        Retorna uma lista de nomes de disciplinas únicas presentes no banco de questões
        para um determinado evento.
        """
        sql = """
            SELECT DISTINCT disciplina_banco 
            FROM QuestoesBanco 
            WHERE id_evento_fk = ? AND disciplina_banco IS NOT NULL AND disciplina_banco != ''
            ORDER BY disciplina_banco;
        """
        rows = self._execute_query(sql, (event_id,), fetch_all=True)
        return [row['disciplina_banco'] for row in rows] if rows else []


    def get_all_disciplinas_in_event_question_bank(self, event_id: str) -> List[Dict[str, Any]]:
        """
        Retorna uma lista de dicionários com informações das disciplinas
        (nome e contagem de questões) para um determinado evento.
        """
        sql = """
            SELECT 
                disciplina_banco, 
                COUNT(id_questao) as quantidade_questoes
            FROM QuestoesBanco
            WHERE id_evento_fk = ? AND disciplina_banco IS NOT NULL AND disciplina_banco != ''
            GROUP BY disciplina_banco
            ORDER BY disciplina_banco;
        """
        rows = self._execute_query(sql, (event_id,), fetch_all=True)
        return [dict(row) for row in rows] if rows else []
