def _selecionar_questoes_com_balanceamento(self, pool_de_questoes: List[QuestaoType], qtd_a_selecionar: int) -> List[QuestaoType]:
        """
        Seleciona um número de questões de um pool, garantindo balanceamento
        e um embaralhamento intercalado para evitar padrões e repetições.
        """
        if not pool_de_questoes or qtd_a_selecionar <= 0:
            return []

        # 1. Separa as questões em listas, uma para cada gabarito.
        questoes_por_gabarito = {'A': [], 'B': [], 'C': [], 'D': [], 'E': []}
        for q in pool_de_questoes:
            gabarito = getattr(q, 'resposta_correta', '').upper()
            if gabarito in questoes_por_gabarito:
                questoes_por_gabarito[gabarito].append(q)

        # Embaralha cada lista individualmente para garantir aleatoriedade interna
        for key in questoes_por_gabarito:
            random.shuffle(questoes_por_gabarito[key])

        # 2. Lógica do "Sorteio Circular"
        prova_final = []
        opcoes_para_sorteio = ['A', 'B', 'C', 'D', 'E']
        
        # Loop para selecionar cada uma das questões da prova
        while len(prova_final) < qtd_a_selecionar:
            
            # Embaralha a ordem de prioridade dos gabaritos a cada seleção
            random.shuffle(opcoes_para_sorteio)
            
            pegou_nesta_rodada = False
            for opcao in opcoes_para_sorteio:
                # Se o monte (a lista de questões) desta opção ainda tem itens...
                if questoes_por_gabarito[opcao]:
                    # Pega uma questão e a remove do monte para não ser usada de novo
                    questao = questoes_por_gabarito[opcao].pop()
                    prova_final.append(questao)
                    pegou_nesta_rodada = True
                    # Para e vai para a próxima questão da prova (próxima iteração do while)
                    break 
            
            # Condição de segurança: se passamos por todos os montes e não tinha 
            # mais nenhuma questão em nenhum deles, encerramos.
            if not pegou_nesta_rodada:
                self.logger.warning("Não há mais questões disponíveis em nenhum grupo para continuar o sorteio.")
                break

        self.logger.info(f"Seleção final com {len(prova_final)} questões (balanceadas e intercaladas).")
        return prova_final
